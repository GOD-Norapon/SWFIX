        // UUID Generator for Idempotency

        function makeUUIDv4() {

            if (crypto?.randomUUID) return crypto.randomUUID();

            return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {

                var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8);

                return v.toString(16);

            });

        }

                toast.classList.replace('bg-emerald-600', 'bg-red-600');

                icon.className = 'ph ph-x-circle text-xl';

            } else {

                toast.classList.replace('bg-red-600', 'bg-emerald-600');

                icon.className = 'ph ph-check-circle text-xl';

                const result = await response.json();

                

                if (result.created !== false) {

                    showToast(result.message || 'บันทึกข้อมูลสำเร็จ');

                    document.getElementById('repairForm').reset();

                    // เปลี่ยนไปหน้ารายการ

                    setTimeout(() => switchTab('list'), 1000);

                } else {

                    showToast(result.message, 'error');

                }

            } catch (error) {

                showToast('เกิดข้อผิดพลาดในการเชื่อมต่อ', 'error');

                console.error(error);

            } finally {

                isSubmitting = false;

                btn.innerHTML = originalText;

                btn.disabled = false;

                btn.classList.remove('opacity-70', 'cursor-not-allowed');

            }

        });


        // Load List Data

        async function loadListData(page = currentPage) {

            if (SCRIPT_URL.includes('YOUR_URL_HERE')) return;

            const tbody = document.getElementById('tableBody');

            tbody.innerHTML = '<tr><td colspan="6" class="text-center py-10 text-slate-500"><i class="ph ph-spinner animate-spin text-2xl mx-auto mb-2"></i>กำลังโหลดข้อมูล...</td></tr>';

            

            try {

                const response = await fetch(`${SCRIPT_URL}?action=getList&page=${page}`);

                const res = await response.json();

                

                currentPage = res.currentPage;

                

                document.getElementById('pageInfo').textContent = `หน้าที่ ${res.currentPage} จาก ${res.totalPages}`;

                document.getElementById('prevBtn').disabled = !res.hasPreviousPage;

                document.getElementById('nextBtn').disabled = !res.hasNextPage;


                renderTable(res.data);

            } catch (error) {

                tbody.innerHTML = '<tr><td colspan="6" class="text-center py-10 text-red-500">ไม่สามารถดึงข้อมูลได้ โปรดลองอีกครั้ง</td></tr>';

            }

        }


        function changePage(delta) {

            loadListData(currentPage + delta);

        }


        function getStatusBadge(status) {

            const s = String(status || '').trim();

            if (s === 'รอตรวจสอบ' || s === '') return `<span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-amber-100 text-amber-800"><span class="w-1.5 h-1.5 bg-amber-500 rounded-full mr-1.5"></span>รอตรวจสอบ</span>`;

            if (s === 'ซ่อมแล้ว') return `<span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-emerald-100 text-emerald-800"><span class="w-1.5 h-1.5 bg-emerald-500 rounded-full mr-1.5"></span>ซ่อมแล้ว</span>`;

            if (s === 'ซ่อมไม่ได้') return `<span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-red-100 text-red-800"><span class="w-1.5 h-1.5 bg-red-500 rounded-full mr-1.5"></span>ซ่อมไม่ได้</span>`;

            if (s === 'รออะไหล่') return `<span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-blue-100 text-blue-800"><span class="w-1.5 h-1.5 bg-blue-500 rounded-full mr-1.5"></span>รออะไหล่</span>`;

            return `<span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-slate-100 text-slate-800">${s}</span>`;

        }


        function renderTable(data) {

            const tbody = document.getElementById('tableBody');

            if (!data || data.length === 0) {

                tbody.innerHTML = '<tr><td colspan="6" class="text-center py-10 text-slate-500">ยังไม่มีข้อมูลแจ้งซ่อม</td></tr>';

                return;

            }


            tbody.innerHTML = data.map(row => `

                <tr class="hover:bg-slate-50 transition-colors group">

                    <td data-label="วันที่แจ้ง" class="px-6 py-4 text-slate-500">${row[0] || '-'}</td>

                    <td data-label="ลำดับ" class="px-6 py-4 text-center font-medium text-slate-900 md:text-slate-500">${row[1] || '-'}</td>

                    <td data-label="ชื่อผู้แจ้ง" class="px-6 py-4 font-medium text-slate-900">${row[2] || '-'}</td>

                    <td data-label="รายการ" class="px-6 py-4 text-slate-600">${row[3] || '-'}</td>

                    <td data-label="สถานที่" class="px-6 py-4 text-slate-600">${row[4] || '-'} / ห้อง ${row[5] || '-'}</td>

                    <td data-label="สถานะ" class="px-6 py-4">${getStatusBadge(row[6])}</td>

                </tr>

            `).join('');

        }

        async function loadStatsData() {

            if (SCRIPT_URL.includes('YOUR_URL_HERE')) return;

            try {

                const response = await fetch(`${SCRIPT_URL}?action=getStats`);

            } catch (error) {

            }

        }


        function renderChart(stats) {

            const ctx = document.getElementById('statusChart').getContext('2d');

            

            if (chartInstance) {

                chartInstance.destroy();

            }


            const dataValues = [stats.pending, stats.completed, stats.failed, stats.waiting];

            // ตรวจสอบว่ามีข้อมูลหรือไม่

            const hasData = dataValues.some(val => val > 0);


            chartInstance = new Chart(ctx, {

                type: 'doughnut',

                data: {

                    labels: ['รอตรวจสอบ', 'ซ่อมเสร็จแล้ว', 'ซ่อมไม่ได้', 'รออะไหล่'],

                    datasets: [{

                        data: hasData ? dataValues : [1],

                        backgroundColor: hasData 

                            ? ['#fbbf24', '#34d399', '#f87171', '#60a5fa'] 

                            : ['#f1f5f9'], // สีเทาถ้าไม่มีข้อมูล

                        borderWidth: 0,

                        hoverOffset: 4

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    cutout: '70%',

                    plugins: {

                        legend: { position: 'bottom', labels: { font: { family: "'Prompt', sans-serif" } } },

                        tooltip: { 

                            enabled: hasData,

                            titleFont: { family: "'Prompt', sans-serif" },

                            bodyFont: { family: "'Prompt', sans-serif" }

                        }

                    }

                }

            });

        }
